package codexfrontend

import (
	"os"
	"os/exec"
	"strconv"
	"strings"
	"testing"
	"time"

	"github.com/ivo-lopes/ivoai/internal/session"
)

// This initial diagnostic helper retains the legacy two-read semantics so the
// regression can demonstrate its failure before the single-snapshot correction.
func nativeFixtureWriterAlive(identity string, readStat func() ([]byte, error)) bool {
	initial, err := readStat()
	if err != nil {
		return false
	}
	end := strings.LastIndexByte(string(initial), ')')
	if end < 0 {
		return false
	}
	fields := strings.Fields(string(initial[end+1:]))
	if len(fields) < 20 {
		return false
	}
	if _, err := strconv.ParseUint(fields[19], 10, 64); err != nil || fields[19] != identity {
		return false
	}
	current, err := readStat()
	if err != nil {
		return true
	}
	end = strings.LastIndexByte(string(current), ')')
	if end < 0 {
		return true
	}
	fields = strings.Fields(string(current[end+1:]))
	return len(fields) == 0 || fields[0] != "Z" && fields[0] != "X"
}

func TestNativeFixtureWriterCannotReviveAfterReap(t *testing.T) {
	cmd := exec.Command("/bin/sh", "-c", "read ignored; exit 0")
	cmd.Env = []string{"PATH=/usr/bin:/bin", "HOME=" + t.TempDir(), "LANG=C"}
	input, err := cmd.StdinPipe()
	if err != nil {
		t.Fatal(err)
	}
	if err := cmd.Start(); err != nil {
		t.Fatal(err)
	}
	waited := false
	defer func() {
		input.Close()
		if !waited {
			cmd.Process.Kill()
			cmd.Wait()
		}
	}()
	identity := session.ProcessStart(cmd.Process.Pid)
	if identity == "" {
		t.Fatal("owned child identity unavailable")
	}
	input.Close()
	path := "/proc/" + strconv.Itoa(cmd.Process.Pid) + "/stat"
	zombie := false
	for i := 0; i < 100; i++ {
		body, err := os.ReadFile(path)
		if err == nil {
			end := strings.LastIndexByte(string(body), ')')
			if end >= 0 {
				fields := strings.Fields(string(body[end+1:]))
				if len(fields) >= 20 && fields[0] == "Z" {
					zombie = true
					break
				}
			}
		}
		time.Sleep(5 * time.Millisecond)
	}
	if !zombie {
		t.Fatal("owned child did not reach controlled zombie state")
	}
	reads := 0
	readThenReap := func() ([]byte, error) {
		body, err := os.ReadFile(path)
		reads++
		if !waited {
			if err := cmd.Wait(); err != nil {
				t.Fatal("controlled owned child exit was not zero")
			}
			waited = true
		}
		return body, err
	}
	if nativeFixtureWriterAlive(identity, readThenReap) {
		t.Fatal("known terminated writer became alive after owned reap between observations")
	}
	t.Logf("owned zero-exit child remains terminated after reap; proc snapshots=%d", reads)
}
