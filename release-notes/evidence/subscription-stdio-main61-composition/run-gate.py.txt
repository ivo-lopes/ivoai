import hashlib, json, pathlib, subprocess, sys, time

proof = pathlib.Path(__file__).resolve().parent
root = pathlib.Path('/var/tmp/ivoai-subscription-stdio-main61-composition-20261029')
go = '/var/tmp/ivoai-validation-toolchain-20261028/go/bin/go'
commands = {
    'full-go': [go, 'test', '-count=1', '-json', './...'],
    'Python': ['/usr/bin/python3', '-m', 'unittest', 'discover', '-s', 'scripts', '-p', 'test_benchmark*.py'],
    'vet': [go, 'vet', './...'],
}
name = sys.argv[1]
command = commands[name]
frozen = json.loads((proof / 'source-frozen.json').read_text())

def verify():
    for path, digest in frozen['source_files'].items():
        assert hashlib.sha256((root / path).read_bytes()).hexdigest() == digest, path

verify()
home = proof / (name + '-home')
home.mkdir(mode=0o700)
environment = {'HOME': str(home), 'PATH': '/var/tmp/ivoai-validation-toolchain-20261028/go/bin:/usr/bin:/bin',
               'GOENV': 'off', 'GOWORK': 'off', 'GOPROXY': 'off', 'GOSUMDB': 'off', 'GOMAXPROCS': '2',
               'GOCACHE': '/home/ivo/.cache/go-build', 'GOMODCACHE': '/home/ivo/go/pkg/mod',
               'TMPDIR': str(proof), 'GOTMPDIR': str(proof), 'LANG': 'C.UTF-8',
               'PYTHONPATH': 'scripts', 'PYTHONDONTWRITEBYTECODE': '1'}
(proof / (name + '.started.json')).open('x').write(json.dumps({'source': frozen['source'], 'command': command, 'source_freeze_sha256': hashlib.sha256((proof / 'source-frozen.json').read_bytes()).hexdigest(), 'once': True, 'environment': environment, 'all_official_client_optins': 'ABSENT'}, indent=2) + '\n')
start = time.monotonic()
with (proof / (name + '.log')).open('xb') as out, (proof / (name + '.stderr')).open('xb') as err:
    result = subprocess.run(command, cwd=root, env=environment, stdout=out, stderr=err)
verify()
record = {'source': frozen['source'], 'gate': name, 'exit': result.returncode,
          'wall_seconds': round(time.monotonic() - start, 3), 'source_unchanged_after': True,
          'official_client_auth_model_cohort_or_service_calls': 0, 'comparative_binary_pair_rebuilt': False}
(proof / (name + '.result.json')).write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps(record))
raise SystemExit(result.returncode)
