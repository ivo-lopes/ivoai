import hashlib, json, os, pathlib, subprocess, sys, time

proof = pathlib.Path(__file__).resolve().parent
root = pathlib.Path('/var/tmp/ivoai-v015-publication-record-20261029')
name = sys.argv[1]
commands = {
    'docs-build-final': ['/bin/bash', 'scripts/build-docs.sh'],
    'docs-check-final': ['/bin/bash', 'scripts/build-docs.sh', '--check'],
    'docsserver-final': ['/var/tmp/ivoai-validation-toolchain-20261028/go/bin/go', 'test', '-count=1', '-json', './internal/docsserver'],
}
assert name in commands
frozen = json.loads((proof / 'immutable-before.json').read_text())
documents = json.loads((proof / 'docs-after-final.json').read_text())

def verify():
    for path, digest in {**frozen['files'], **frozen['locked_docs_inputs'], **documents}.items():
        file = root / path
        body = os.readlink(file).encode() if file.is_symlink() else file.read_bytes()
        assert hashlib.sha256(body).hexdigest() == digest, path

verify()
environment = json.loads((proof / 'environment-network-metadata.json').read_text())
(proof / (name + '.started.json')).open('x').write(json.dumps({'base': frozen['source'], 'command': commands[name], 'documents': documents,
    'environment': environment, 'once': True, 'package_metadata_network_allowed': True}, indent=2) + '\n')
start = time.monotonic()
with (proof / (name + '.log')).open('xb') as out, (proof / (name + '.stderr')).open('xb') as err:
    result = subprocess.run(commands[name], cwd=root, env=environment, stdout=out, stderr=err)
verify()
record = {'base': frozen['source'], 'gate': name, 'exit': result.returncode, 'wall_seconds': round(time.monotonic() - start, 3),
    'immutable_source_and_records_unchanged_after': True, 'document_inputs_unchanged_after': True,
    'official_client_auth_model_or_production_service_calls': 0, 'dependencies': 'own frozen lock/integrity; normal public package metadata allowed; no dependency/version change'}
(proof / (name + '.result.json')).write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps(record))
raise SystemExit(result.returncode)
