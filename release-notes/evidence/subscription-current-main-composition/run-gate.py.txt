import hashlib, json, os, pathlib, subprocess, sys, time

proof = pathlib.Path(__file__).resolve().parent
root = pathlib.Path('/var/tmp/ivoai-subscription-current-main-composition-20261029')
go = '/var/tmp/ivoai-validation-toolchain-20261028/go/bin/go'
commands = {
    'App-hermetic': [go, 'test', '-count=1', '-json', './internal/app', '-run', '^TestPrimaryPrefetch(DirectMCP.*|Subscription.*)$'],
    'server-docs': [go, 'test', '-count=1', '-json', './internal/servercmd', './internal/docsserver'],
    'cmd-compile': [go, 'build', '-trimpath', '-o', str(proof / 'ivoai-linux-amd64v1'), './cmd/ivoai'],
}
name = sys.argv[1]
command = commands[name]
frozen = json.loads((proof / 'source-frozen.json').read_text())

def verify():
    for path, record in frozen['files'].items():
        file = root / path
        assert file.is_symlink() == (record['kind'] == 'symlink'), path
        body = os.readlink(file).encode() if file.is_symlink() else file.read_bytes()
        assert hashlib.sha256(body).hexdigest() == record['sha256'], path

verify()
home = proof / (name + '-home')
home.mkdir(mode=0o700)
environment = {'HOME': str(home), 'PATH': '/var/tmp/ivoai-validation-toolchain-20261028/go/bin:/usr/bin:/bin',
               'GOENV': 'off', 'GOWORK': 'off', 'GOPROXY': 'off', 'GOSUMDB': 'off', 'GOMAXPROCS': '2',
               'GOCACHE': '/home/ivo/.cache/go-build', 'GOMODCACHE': '/home/ivo/go/pkg/mod',
               'TMPDIR': str(proof), 'GOTMPDIR': str(proof), 'LANG': 'C.UTF-8'}
if name == 'cmd-compile':
    environment.update(CGO_ENABLED='0', GOOS='linux', GOARCH='amd64', GOAMD64='v1')
(proof / (name + '.started.json')).open('x').write(json.dumps({'source': frozen['source'], 'source_freeze_sha256': hashlib.sha256((proof / 'source-frozen.json').read_bytes()).hexdigest(), 'command': command, 'environment': environment, 'once': True, 'all_official_optins': 'ABSENT'}, indent=2) + '\n')
start = time.monotonic()
with (proof / (name + '.log')).open('xb') as out, (proof / (name + '.stderr')).open('xb') as err:
    result = subprocess.run(command, cwd=root, env=environment, stdout=out, stderr=err)
verify()
record = {'source': frozen['source'], 'gate': name, 'exit': result.returncode, 'wall_seconds': round(time.monotonic() - start, 3),
          'all_frozen_paths_unchanged_after': len(frozen['files']), 'official_client_auth_model_or_deployed_service_calls': 0,
          'full_Go_race_Python_Node_run': False, 'comparative_binary_pair_rebuilt': False}
(proof / (name + '.result.json')).write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps(record))
raise SystemExit(result.returncode)
