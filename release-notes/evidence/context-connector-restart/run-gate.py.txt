import hashlib, json, pathlib, subprocess, sys, time

proof = pathlib.Path(__file__).resolve().parent
root = pathlib.Path('/var/tmp/ivoai-context-connector-restart-20261029')
go = '/var/tmp/ivoai-validation-toolchain-20261028/go/bin/go'
commands = {
    'focused-race': [go, 'test', '-race', '-count=1', '-timeout', '120s', '-json', './internal/servercmd', './internal/server', './internal/context'],
    'vet': [go, 'vet', './internal/servercmd', './internal/server', './internal/context'],
    'linux-arm64-compile': [go, 'test', '-c', '-o', str(proof / 'servercmd-linux-arm64.test'), './internal/servercmd'],
}
name = sys.argv[1]
command = commands[name]
frozen = json.loads((proof / 'source-frozen.json').read_text())

def verify():
    for path, digest in frozen['source_files'].items():
        assert hashlib.sha256((root / path).read_bytes()).hexdigest() == digest, path

verify()
(proof / (name + '.started.json')).open('x').write(json.dumps({'source': frozen['source_commit'], 'command': command, 'once': True}, indent=2) + '\n')
environment = {'HOME': str(proof), 'PATH': '/var/tmp/ivoai-validation-toolchain-20261028/go/bin:/usr/bin:/bin',
               'GOENV': 'off', 'GOWORK': 'off', 'GOPROXY': 'off', 'GOSUMDB': 'off', 'GOMAXPROCS': '2',
               'GOCACHE': '/home/ivo/.cache/go-build', 'GOMODCACHE': '/home/ivo/go/pkg/mod',
               'TMPDIR': str(proof), 'GOTMPDIR': str(proof), 'LANG': 'C.UTF-8'}
if name == 'linux-arm64-compile':
    environment.update(GOOS='linux', GOARCH='arm64', CGO_ENABLED='0')
start = time.monotonic()
with (proof / (name + '.log')).open('xb') as out, (proof / (name + '.stderr')).open('xb') as err:
    result = subprocess.run(command, cwd=root, env=environment, stdout=out, stderr=err)
verify()
record = {'source': frozen['source_commit'], 'gate': name, 'exit': result.returncode,
          'wall_seconds': round(time.monotonic() - start, 3), 'source_unchanged_after': True,
          'official_client_auth_model_or_service_calls': 0, 'external_Context_sync': 'BLOCKED_NOT_EXECUTED'}
(proof / (name + '.result.json')).write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps(record))
raise SystemExit(result.returncode)
