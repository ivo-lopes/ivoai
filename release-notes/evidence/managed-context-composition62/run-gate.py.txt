import datetime,hashlib,json,pathlib,subprocess,sys,time
proof=pathlib.Path(__file__).resolve().parent
source=pathlib.Path('/var/tmp/ivoai-mcp-gateway-composition-20261029')
name=sys.argv[1]
commands={'full-go':['test','-count=1','-json','./...'],'vet':['vet','./...']}
assert name in commands
freeze=json.loads((proof/'source-frozen.json').read_text())
def verify():
 assert all(hashlib.sha256((source/n).read_bytes()).hexdigest()==h for n,h in freeze['files'].items())
verify()
with (proof/(name+'.started.json')).open('x') as f:
 json.dump({'source':freeze['source_commit'],'source_freeze_sha256':hashlib.sha256((proof/'source-frozen.json').read_bytes()).hexdigest(),'max_runs':1,'command':['go']+commands[name],'started_utc':datetime.datetime.now(datetime.timezone.utc).isoformat()},f,indent=2)
command=['/var/tmp/ivoai-validation-toolchain-20261028/go/bin/go']+commands[name]
env={'HOME':str(proof),'PATH':'/var/tmp/ivoai-validation-toolchain-20261028/go/bin:/usr/bin:/bin','GOCACHE':'/home/ivo/.cache/go-build','GOMODCACHE':'/home/ivo/go/pkg/mod','GOENV':'off','GOWORK':'off','GOPROXY':'off','GOSUMDB':'off','LANG':'C.UTF-8'}
start=time.monotonic()
with (proof/(name+('.jsonl' if name=='full-go' else '.log'))).open('xb') as out,(proof/(name+'.stderr')).open('xb') as err:
 result=subprocess.run(command,cwd=source,env=env,stdout=out,stderr=err)
verify()
record={'source':freeze['source_commit'],'exit':result.returncode,'count':1,'wall_seconds':round(time.monotonic()-start,3),'finished_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'Go_source_unchanged_after':True,'no_official_cli_auth_status_provider':True}
(proof/(name+'.result.json')).write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps(record));sys.exit(result.returncode)
